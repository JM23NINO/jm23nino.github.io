module Sumador3BitCLA (
 input [2:0] A, //todas la variables de 3 bits
 input [2:0] B,
 input Cin,
 output [2:0] Sum,
 output Cout
);

 wire [2:0] C;
 wire [2:0] P;

 wire C1;
 wire C2;
 assign C[0] = A[0] & B[0];
 assign C[1] = A[1] & B[1];
 assign C[2] = A[2] & B[2];
 assign P[0] = A[0] ^ B[0];
 assign P[1] = A[1] ^ B[1];
 assign P[2] = A[2] ^ B[2];
 assign C1 = C[0] | (P[0] & Cin);
 assign C2 = C[1] | (P[1] & C[0]) | (P[1] & P[0] & Cin);
 assign Cout = C[2] | (P[2] & C[1]) | (P[2] & P[1] & C[0]) | (P[2] & P[1] & P[0] & Cin);
 assign Sum[0] = P[0] ^ Cin;
 assign Sum[1] = P[1] ^ C1;
 assign Sum[2] = P[2] ^ C2;
endmodule
